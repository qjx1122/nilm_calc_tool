# nilm_calc_tool — 运行依赖
# 安装：python -m venv .venv && .venv/bin/pip install -r requirements.txt
#      （Windows：py -m venv .venv && .venv\Scripts\pip install -r requirements.txt）
# 说明：算法核心仅依赖 numpy；Web 界面依赖 fastapi + uvicorn；两者可分开安装（CLI 模式无需 Web 依赖）

# --- 算法核心（必需） ---

# --- Web 界面（子界面导航 + 实时推送，可选但推荐） ---

# 在 Windows 上（尤其是 `pip download --platform win_amd64` 离线打包）会直接
# ResolutionImpossible。故按平台分流：Windows 用纯 uvicorn + websockets（WebSocket 推送够用）。
uvicorn[standard]>=0.27; sys_platform != "win32"
uvicorn>=0.27; sys_platform == "win32"
websockets>=12; sys_platform == "win32"

# --- 开发 / 验证（可选） ---
pytest>=8.0
httpx>=0.27        # FastAPI TestClient 依赖（Web 层端到端测试 tests/test_server.py）
